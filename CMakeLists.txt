cmake_minimum_required(VERSION 3.25)

project(echo_mini_fw LANGUAGES C)

if(WIN32)
    enable_language(RC)
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(ECHO_SANITIZE "Build host code with UndefinedBehaviorSanitizer" OFF)
option(ECHO_ADDRESS_SANITIZER "Build host code with AddressSanitizer" OFF)
option(ECHO_BUILD_SIM "Build the SDL2 simulator" ON)
option(ECHO_BUILD_TESTS "Build the host tests" ON)
option(ECHO_BUILD_DEVICE "Build the device firmware with cmake/arm-none-eabi.cmake" OFF)
option(ECHO_RELEASE "Leave the developer-only Diagnostics rows out of the build" OFF)
option(ECHO_JACK_DIAGNOSTICS "Expose the wired-jack diagnostic menu and transition counters" OFF)
if(ECHO_JACK_DIAGNOSTICS)
    add_compile_definitions(ECHO_JACK_DIAGNOSTICS=1)
endif()
set(ECHO_BOARD "echo-mini" CACHE STRING "Board profile: echo-mini, neo or neo-sim")
set_property(CACHE ECHO_BOARD PROPERTY STRINGS echo-mini neo neo-sim)
if(ECHO_BOARD STREQUAL "neo-sim")
    if(ECHO_BUILD_DEVICE)
        message(FATAL_ERROR "Neo hardware drivers are not verified. neo-sim is a desktop-only target.")
    endif()
    add_compile_definitions(ECHO_NEO_SIM=1)
elseif(ECHO_BOARD STREQUAL "neo")
    add_compile_definitions(ECHO_BOARD_NEO=1)
elseif(NOT ECHO_BOARD STREQUAL "echo-mini")
    message(FATAL_ERROR "Unknown ECHO_BOARD: ${ECHO_BOARD}")
endif()
set(ECHO_APP_RAM_BUDGET_BYTES 131072 CACHE STRING "Static RAM budget for src/ on the device")

include(cmake/project_options.cmake)

add_subdirectory(src)
add_subdirectory(platform/device/logic)

if(ECHO_BUILD_SIM)
    add_subdirectory(platform/sim)
endif()

if(ECHO_BUILD_DEVICE)
    add_subdirectory(platform/device)
endif()

if(ECHO_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
